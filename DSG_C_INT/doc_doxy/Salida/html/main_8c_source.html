134void generate_waveform() {
135 static uint64_t last_time = 0;
136 uint64_t current_time = time_us_64();
137 float value = 0;
138
139 // Calcular el paso de tiempo
140 float time_step = (current_time - last_time) / 1000000.0f;
141 last_time = current_time;
142
143 switch (current_waveform) {
144 case SINE:
145 value = amplitude * (sinf(2 * M_PI * frequency * time_step) / 2 + 0.5) + dc_offset;
146 break;
147 case SQUARE:
148 value = amplitude * (sinf(2 * M_PI * frequency * time_step) > 0 ? 1 : -1) / 2 + 0.5 + dc_offset;
149 break;
150 case SAWTOOTH:
151 value = amplitude * (2 * fmod(frequency * time_step, 1) - 1) / 2 + 0.5 + dc_offset;
152 break;
153 case TRIANGULAR:
154 value = amplitude * (1.0 - fabs(fmod(frequency * time_step * 4, 4) - 2)) + dc_offset;
155 break;
156 }
157
158 // Convertir a valor DAC
159 uint dac_value = (uint)((value / VREF) * DAC_MAX_VALUE);
160 gpio_put(DAC_PIN, dac_value); // Suponiendo que DAC_PIN estÃ¡ conectado a un DAC de escalera R2R simple o similar
161}
